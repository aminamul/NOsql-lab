      -[r:WORKS_ON]->
      (p:Project {id: 101})

       r.role AS Role,
       p.title AS Project;

MATCH (p:Project)
WHERE p.budget >= 1000000
RETURN p.title AS Project,
       p.budget AS Budget
ORDER BY Budget DESC;

MATCH (d:Developer)
WHERE d.department = "Backend"
RETURN d.name AS Developer,
       d.department AS Department;

MATCH (d:Developer)-[:WORKS_ON]->(p:Project)-[:USES]->(t:Technology)
RETURN d.name AS Developer,
       p.title AS Project,
       t.name AS Technology
ORDER BY Developer;

MATCH (p:Project)<-[:WORKS_ON]-(d:Developer)
WHERE p.title = "Online Store"
RETURN p.title AS Project,
       d.name AS Developer;

MATCH (d:Developer)-[:WORKS_ON]->(p:Project)
RETURN d.name AS Developer,
       p.title AS Project;

MATCH (d:Developer)
RETURN d.name AS Developer,
       d.level AS Level,
       d.department AS Department
ORDER BY Developer
LIMIT 10;
